# dotfiles-Kali :: base CLI stack (apt / Debian names)
# One package per line. '#' comments and blank lines are ignored by bootstrap.sh.
# Note: on Debian/Kali, `bat` ships the binary `batcat` and `fd-find` ships
# `fdfind` — Core's 00-tools.zsh already resolves those alt binary names.

# shell + multiplexer
zsh
tmux
starship       # prompt — Kali packages it, so take the distro build rather than
               # piping starship.rs/install.sh into sh. The other out-of-band tools
               # (atuin/mise/uv/ty) are NOT packaged and come from pinned, sha256-
               # verified release assets instead: see install/tool-versions.env.

# modern coreutils replacements
eza
bat            # binary: batcat
fd-find        # binary: fdfind
ripgrep
zoxide
fzf
git-delta      # binary: delta
btop
tealdeer       # tldr client (binary: tldr); swap to `tldr` if unavailable

# the rest of core-doctor's modern-CLI set (apt/Debian names)
du-dust        # binary: dust — the package is `du-dust`, the command stays `dust`
duf            # disk-usage/free viewer
procs          # modern `ps`
sd             # find & replace (src: rust-sd); binary: sd
gron           # greppable JSON
xh             # HTTPie-style HTTP client (src: rust-xh); binary: xh
yq-go          # mikefarah's Go `yq` — Debian ALSO ships `yq` (Python kislyuk);
               # we want the Go one, so use `yq-go` and NOT plain `yq`.
glow           # markdown renderer — recent (Debian sid → Kali rolling); may lag migration
gum            # charmbracelet TUI toolkit — recent (Debian sid → Kali rolling); may lag migration
difftastic     # structural git diff (binary: difft) — Core wires `gdft` when HAVE_DIFFT; opt-in,
               # recent (Debian sid → Kali rolling); may lag migration. (ast-grep, its search
               # counterpart, isn't in apt — bootstrap.sh cargo-installs it, like viddy.)
# core-doctor also wants: doggo, sesh (go), carapace (upstream .deb), op (vendor apt
# repo) — all installed by bootstrap.sh (not in apt / not reliably packaged), not listed
# here. carapace is deliberately NOT go-installed: `go install` cannot work for any
# version of it (replace directives + uncommitted generated sources). See bootstrap.sh's
# carapace block, and core/PORTING-MATRIX.md's carapace footnote (²⁷ there — it reaches
# this vendored copy with the next Core sync; dotgibson/dotfiles-core#468).

# Kali-only modern CLI — deliberately NOT in Core (dotfiles-core#395 routed it here)
hexyl          # hex viewer — the stack's only binary-inspection verb. Own command, no alias
               # (shadows nothing classic); offensive.zsh sets HAVE_HEXYL. Core's tool scan
               # (dotfiles-core#376) rated it adopt-if-you-want: low daily frequency and
               # "inspect a binary" is an operator verb, not a fleet-wide one.

# editor + vcs
neovim
git
lazygit        # git TUI — Core ships `alias lg='lazygit'` and a `prefix + g` tmux popup

               # and the popup are dead on Kali alone. Every other Linux repo installs it.
openssh-client
git-credential-manager   # optional; remove if not packaged on your box

# clipboard (mostly inert on WSL — Core uses clip.exe — but harmless)

# everyday utilities
curl
wget
jq
gnupg         # gpg — an UNDECLARED dependency until now: bootstrap.sh's `op` step
              # dearmors 1Password's signing key into /usr/share/keyrings, and a
              # minimal image (the Kali WSL rootfs among them) does not ship gpg.
              # Declaring it does not fix the `sudo: gpg: command not found` case —
              # that one is sudo's secure_path, not a missing package — but an
              # undeclared hard dependency should be listed regardless, and any
              # future signed-apt-repo step needs it too.
w3m            # terminal web browser (text-mode; renders tables + inline images)
gawk
direnv
unzip
